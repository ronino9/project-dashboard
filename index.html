// C/S 리스크맵 함수들 function renderCsBubbleChart() { const canvas = document.getElementById('csBubbleCanvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); const rect = canvas.getBoundingClientRect(); canvas.width = rect.width; canvas.height = rect.height; const padding = 60; const graphWidth = canvas.width - padding * 2; const graphHeight = canvas.height - padding * 2; ctx.fillStyle = '#0f172a'; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.strokeStyle = '#334155'; ctx.lineWidth = 1; for (let i = 0; i <= 5; i++) { const x = padding + (graphWidth / 5) * i; const y = padding + (graphHeight / 5) * i; ctx.beginPath(); ctx.moveTo(x, padding); ctx.lineTo(x, canvas.height - padding); ctx.stroke(); ctx.beginPath(); ctx.moveTo(padding, y); ctx.lineTo(canvas.width - padding, y); ctx.stroke(); } ctx.strokeStyle = '#64748b'; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(padding, canvas.height - padding); ctx.lineTo(canvas.width - padding, canvas.height - padding); ctx.stroke(); ctx.beginPath(); ctx.moveTo(padding, padding); ctx.lineTo(padding, canvas.height - padding); ctx.stroke(); ctx.fillStyle = '#94a3b8'; ctx.font = '12px sans-serif'; ctx.textAlign = 'center'; ctx.fillText('← 임박도 →', canvas.width / 2, canvas.height - 10); if (csRiskData.length === 0) { ctx.fillStyle = '#64748b'; ctx.font = '14px sans-serif'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText('데이터 없음', canvas.width / 2, canvas.height / 2); return; } let maxX = Math.max(...csRiskData.map(p => p.x), 1); let maxY = Math.max(...csRiskData.map(p => p.y), 1); const colorMap = { '안정': '#86EFAC', '임박': '#FCD34D', '주의': '#F9A8D4', '위험': '#FCA5A5', '심각': '#EF4444' }; csRiskData.forEach(project => { const x = padding + (project.x / maxX) * graphWidth; const y = canvas.height - padding - (project.y / maxY) * graphHeight; const radius = (project.size / 100) * 20 + 15; let color = colorMap[project.status] || '#86EFAC'; ctx.fillStyle = color; ctx.globalAlpha = 0.7; ctx.beginPath(); ctx.arc(x, y, radius, 0, Math.PI * 2); ctx.fill(); ctx.globalAlpha = 1; ctx.strokeStyle = color; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(x, y, radius, 0, Math.PI * 2); ctx.stroke(); ctx.fillStyle = '#f8fafc'; ctx.font = 'bold 13px monospace'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText(project.name, x, y); project.x_pixel = x; project.y_pixel = y; project.radius = radius; }); canvas.onclick = (e) => { const rect = canvas.getBoundingClientRect(); const mouseX = e.clientX - rect.left; const mouseY = e.clientY - rect.top; csRiskData.forEach(project => { const dist = Math.sqrt(Math.pow(mouseX - project.x_pixel, 2) + Math.pow(mouseY - project.y_pixel, 2)); if (dist < project.radius) { selectCsProject(project); } }); }; } function selectCsProject(project) { const html = `
프로젝트명
${project.name}
상태
${project.status}
정보
진행율: ${project.size}%
`; document.getElementById('csDetailContent').innerHTML = html; } // syncDashboard 함수 수정 async function syncDashboard() { try { const response = await fetch(API_URL, { redirect: 'follow' }); allData = await response.json(); // C/S 데이터 추출 if (allData.project1CsRiskMap) { csRiskData = allData.project1CsRiskMap; } switchProject('project7'); } catch (err) { console.error('동기화 실패:', err); } }
