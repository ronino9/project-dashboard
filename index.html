function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('No.7_상세이력'); const range = sheet.getDataRange(); const values = range.getValues(); const headers = values[0]; const data = values.slice(1); const result = { headers: headers, rows: data, count: data.length }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON) .addHeader('Access-Control-Allow-Origin', '*') .addHeader('Access-Control-Allow-Methods', 'GET') .addHeader('Access-Control-Allow-Headers', 'Content-Type'); }
